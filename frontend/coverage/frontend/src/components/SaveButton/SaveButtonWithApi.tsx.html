import type React from 'react'
import { useState } from 'react'
import { useSelector } from 'react-redux'
 
import { useAuth, useDesigns, useGoogleAuth } from '@/hooks/useApi'
import { useNotification } from '@/hooks/useNotification'
import type { IRootState } from '@/store/store'
import Button from '@/components/Button'
import NotificationModal from '@/components/NotificationModal'
 
interface ISaveButtonWithApiProps {
  onSave?: () => void
  loading?: boolean
}
 
const SaveButtonWithApi: React.FC<ISaveButtonWithApiProps> = ({ onSave, loading = false }) => {
  // Random design name generators
  const adjectives = [
    'Awesome',
    'Beautiful',
    'Creative',
    'Dynamic',
    'Elegant',
    'Fresh',
    'Modern',
    'Sleek',
    'Stylish',
    'Vibrant',
  ]
  const nouns = [
    'Design',
    'Layout',
    'Theme',
    'Interface',
    'Style',
    'Template',
    'Concept',
    'Vision',
    'Creation',
    'Masterpiece',
  ]
 
  const generateRandomName = () => {
    const adjective = adjectives[Math.floor(Math.random() * adjectives.length)]
    const noun = nouns[Math.floor(Math.random() * nouns.length)]
    return `${adjective} ${noun}`
  }
 
  const [isAuthModalOpen, setIsAuthModalOpen] = useState(false)
  const [designName, setDesignName] = useState('')
  const [designDescription, setDesignDescription] = useState('')
  const [isSaving, setIsSaving] = useState(false)
  const [sampleRandomName] = useState(() => generateRandomName())
 
  const generateUniqueName = async (baseName: string, existingNames: string[]): Promise<string> => {
    let uniqueName = baseName
    let counter = 1
 
    while (existingNames.includes(uniqueName)) {
      uniqueName = `${baseName} (${counter})`
      counter++
    }
 
    return uniqueName
  }
 
  // Get current design state from Redux
  const {
    selectedUIPreset,
    selectedColorPalette,
    selectedFontSettings,
    selectedSpacingSettings,
    selectedBorderStyle,
    customColors,
  } = useSelector((state: IRootState) => state.designer)
 
  // API hooks
  const { user, isAuthenticated, logout } = useAuth()
  const { designs, createDesign } = useDesigns()
  const { getAuthUrl, handleCallback, loading: authLoading, error: authError } = useGoogleAuth()
  const { notification, showSuccess, showError, showWarning, hideNotification } = useNotification()
 
  const handleSaveClick = () => {
    if (!isAuthenticated) {
      setIsAuthModalOpen(true)
      return
    }
 
    // Check if there's something to save
    if (!selectedUIPreset) {
      showWarning('No Design Selected', 'Please select a UI preset before saving your design.')
      return
    }
 
    handleSaveDesign()
  }
 
  const handleSaveDesign = async () => {
    setIsSaving(true)
 
    try {
      // Generate design name if not provided
      let finalName = designName.trim()
      if (!finalName) {
        finalName = generateRandomName()
      }
 
      // Generate unique name if name already exists
      const existingNames = designs?.map(d => d.name) || []
      const uniqueName = await generateUniqueName(finalName, existingNames)
 
      // Create comprehensive design configuration object
      const designConfig = {
        // UI Structure
        selectedUIPreset,
 
        // Visual Styling
        selectedColorPalette,
        customColors,
        selectedFontSettings,
        selectedSpacingSettings,
        selectedBorderStyle,
 
        // Additional metadata
        createdAt: new Date().toISOString(),
        version: '1.0',
 
        // Include any other design state
        // Add more as your Redux store expands
      }
 
      // Create the design via API
      await createDesign({
        name: uniqueName,
        description: designDescription.trim() || null,
        config: designConfig,
        preset_type: selectedUIPreset?.type || selectedUIPreset?.name || 'custom',
      })
 
      // Success feedback
      showSuccess('Design Saved!', `Your design "${uniqueName}" has been saved successfully.`)
 
      // Reset form
      setDesignName('')
      setDesignDescription('')
      setIsAuthModalOpen(false)
 
      // Call parent onSave if provided
      onSave?.()
    } catch (error) {
      console.error('Failed to save design:', error)
      const errorMessage = error instanceof Error ? error.message : 'An unexpected error occurred'
      showError('Save Failed', `Failed to save your design: ${errorMessage}`)
    } finally {
      setIsSaving(false)
    }
  }
 
  const handleGoogleLogin = async () => {
    try {
      const authUrl = await getAuthUrl()
 
      // Open Google OAuth in a popup
      const popup = window.open(
        authUrl,
        'google-oauth',
        'width=500,height=600,scrollbars=yes,resizable=yes'
      )
 
      // Listen for the OAuth callback
      const messageListener = async (event: MessageEvent) => {
        if (event.origin !== window.location.origin) return
 
        if (event.data.type === 'GOOGLE_OAUTH_SUCCESS') {
          const { code } = event.data
 
          try {
            await handleCallback(code, `${window.location.origin}/auth/callback`)
            popup?.close()
 
            // After successful login, the modal should stay open to show the design form
            // The isAuthenticated state will update and the modal will show the save form
          } catch (error) {
            console.error('OAuth callback failed:', error)
            showError(
              'Authentication Failed',
              'Could not complete Google sign-in. Please try again.'
            )
          }
        }
 
        window.removeEventListener('message', messageListener)
      }
 
      window.addEventListener('message', messageListener)
 
      // Clean up if popup is closed manually
      const checkClosed = setInterval(() => {
        if (popup?.closed) {
          window.removeEventListener('message', messageListener)
          clearInterval(checkClosed)
        }
      }, 1000)
    } catch (error) {
      console.error('Failed to initiate Google login:', error)
      showError('Login Error', 'Failed to start Google authentication. Please try again.')
    }
  }
 
  return (
    <>
      <div className='flex items-center gap-3'>
        {isAuthenticated && user && (
          <div className='flex items-center gap-2 text-sm text-neutral-600 dark:text-neutral-400'>
            <div className='w-8 h-8 bg-primary-500 rounded-full flex items-center justify-center text-white text-xs font-bold'>
              {user.full_name
                ?.split(' ')
                .map((n: string) => n[0])
                .join('') ||
                user.email?.[0]?.toUpperCase() ||
                'U'}
            </div>
            <span>{user.full_name || user.email}</span>
          </div>
        )}
 
        <Button
          variant='primary'
          onClick={handleSaveClick}
          loading={loading || isSaving}
          leftIcon={
            <svg className='w-5 h-5' fill='none' stroke='currentColor' viewBox='0 0 24 24'>
              <path
                strokeLinecap='round'
                strokeLinejoin='round'
                strokeWidth={2}
                d='M8 7H5a2 2 0 00-2 2v9a2 2 0 002 2h14a2 2 0 002-2V9a2 2 0 00-2-2h-3m-1 4l-3-3m0 0l-3 3m3-3v12'
              />
            </svg>
          }
        >
          {isAuthenticated ? 'Save Design' : 'Login & Save'}
        </Button>
 
        {isAuthenticated && (
          <Button variant='secondary' size='sm' onClick={logout}>
            Logout
          </Button>
        )}
      </div>
 
      {/* Auth/Save Modal */}
      {isAuthModalOpen && (
        <div className='fixed inset-0 z-[10000] flex items-center justify-center bg-black bg-opacity-50'>
          <div className='bg-white dark:bg-neutral-800 rounded-2xl p-8 max-w-md w-full mx-4 shadow-2xl'>
            {!isAuthenticated ? (
              // Authentication Step
              <div className='text-center'>
                <div className='text-4xl mb-4'>🎨</div>
                <h2 className='text-2xl font-bold text-neutral-900 dark:text-neutral-100 mb-2'>
                  Save Your Design
                </h2>
                <p className='text-neutral-600 dark:text-neutral-400 mb-6'>
                  Sign in to save and manage your UI designs
                </p>
 
                {authError && (
                  <div className='bg-red-50 dark:bg-red-900/20 border border-red-200 dark:border-red-800 rounded-lg p-3 mb-4'>
                    <p className='text-red-600 dark:text-red-400 text-sm'>{authError}</p>
                  </div>
                )}
 
                <div className='space-y-4'>
                  <Button
                    variant='primary'
                    fullWidth
                    loading={authLoading}
                    onClick={handleGoogleLogin}
                    leftIcon={
                      <svg className='w-5 h-5' viewBox='0 0 24 24'>
                        <path
                          fill='currentColor'
                          d='M22.56 12.25c0-.78-.07-1.53-.2-2.25H12v4.26h5.92c-.26 1.37-1.04 2.53-2.21 3.31v2.77h3.57c2.08-1.92 3.28-4.74 3.28-8.09z'
                        />
                        <path
                          fill='currentColor'
                          d='M12 23c2.97 0 5.46-.98 7.28-2.66l-3.57-2.77c-.98.66-2.23 1.06-3.71 1.06-2.86 0-5.29-1.93-6.16-4.53H2.18v2.84C3.99 20.53 7.7 23 12 23z'
                        />
                        <path
                          fill='currentColor'
                          d='M5.84 14.09c-.22-.66-.35-1.36-.35-2.09s.13-1.43.35-2.09V7.07H2.18C1.43 8.55 1 10.22 1 12s.43 3.45 1.18 4.93l2.85-2.22.81-.62z'
                        />
                        <path
                          fill='currentColor'
                          d='M12 5.38c1.62 0 3.06.56 4.21 1.64l3.15-3.15C17.45 2.09 14.97 1 12 1 7.7 1 3.99 3.47 2.18 7.07l3.66 2.84c.87-2.6 3.3-4.53 6.16-4.53z'
                        />
                      </svg>
                    }
                  >
                    Continue with Google
                  </Button>
 
                  <Button
                    variant='secondary'
                    fullWidth
                    onClick={() => {
                      setIsAuthModalOpen(false)
                    }}
                  >
                    Cancel
                  </Button>
                </div>
 
                <p className='text-xs text-neutral-500 dark:text-neutral-500 mt-4'>
                  By signing in, you agree to our Terms of Service and Privacy Policy
                </p>
              </div>
            ) : (
              // Design Save Step
              <div>
                <h2 className='text-2xl font-bold text-neutral-900 dark:text-neutral-100 mb-4'>
                  Save Design
                </h2>
 
                <div className='space-y-4'>
                  <div>
                    <label className='block text-sm font-medium text-neutral-700 dark:text-neutral-300 mb-2'>
                      Design Name (optional)
                    </label>
                    <input
                      type='text'
                      value={designName}
                      onChange={e => {
                        setDesignName(e.target.value)
                      }}
                      placeholder={`Leave empty for random name (e.g., '${sampleRandomName}')`}
                      className='w-full px-3 py-2 border border-neutral-300 dark:border-neutral-600 rounded-lg focus:ring-2 focus:ring-primary-500 focus:border-transparent bg-white dark:bg-neutral-700 text-neutral-900 dark:text-neutral-100'
                      autoFocus
                    />
                    <p className='text-xs text-neutral-500 dark:text-neutral-400 mt-1'>
                      We'll generate a unique name if you leave this empty. If the name exists,
                      we'll add a number (e.g., "My Design (2)").
                    </p>
                  </div>
 
                  <div>
                    <label className='block text-sm font-medium text-neutral-700 dark:text-neutral-300 mb-2'>
                      Description (optional)
                    </label>
                    <textarea
                      value={designDescription}
                      onChange={e => {
                        setDesignDescription(e.target.value)
                      }}
                      placeholder='Describe your design...'
                      rows={3}
                      className='w-full px-3 py-2 border border-neutral-300 dark:border-neutral-600 rounded-lg focus:ring-2 focus:ring-primary-500 focus:border-transparent bg-white dark:bg-neutral-700 text-neutral-900 dark:text-neutral-100 resize-none'
                    />
                  </div>
 
                  <div className='flex gap-3 pt-4'>
                    <Button
                      variant='primary'
                      onClick={handleSaveDesign}
                      loading={isSaving}
                      fullWidth
                    >
                      Save Design
                    </Button>
                    <Button
                      variant='secondary'
                      onClick={() => {
                        setIsAuthModalOpen(false)
                      }}
                      fullWidth
                    >
                      Cancel
                    </Button>
                  </div>
                </div>
              </div>
            )}
          </div>
        </div>
      )}
 
      {/* Notification Modal */}
      <NotificationModal
        isOpen={notification.isOpen}
        onClose={hideNotification}
        type={notification.type}
        title={notification.title}
        message={notification.message}
        confirmText={notification.confirmText}
        onConfirm={notification.onConfirm}
        showCancel={notification.showCancel}
      />
    </>
  )
}
 
export default SaveButtonWithApi
